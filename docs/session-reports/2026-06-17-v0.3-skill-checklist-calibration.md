from pathlib import Path
import re
import subprocess

skill = Path("skills/evolving-agent-process/SKILL.md").read_text()
match = re.match(r"^---\n(.*?)\n---\n", skill, re.S)
assert match, "frontmatter block missing"

root = {}
stack = [(-1, root)]
for lineno, raw in enumerate(match.group(1).splitlines(), 1):
    if not raw.strip() or raw.lstrip().startswith("#"):
        continue
    indent = len(raw) - len(raw.lstrip(" "))
    assert indent % 2 == 0, f"line {lineno}: indentation is not two-space aligned"
    key, sep, value = raw.strip().partition(":")
    assert sep, f"line {lineno}: missing ':'"
    assert re.fullmatch(r"[A-Za-z0-9_-]+", key), f"line {lineno}: invalid key {key!r}"
    while stack and indent <= stack[-1][0]:
        stack.pop()
    assert stack, f"line {lineno}: invalid nesting"
    parent = stack[-1][1]
    value = value.strip()
    if value == "":
        node = {}
        parent[key] = node
        stack.append((indent, node))
    elif value.startswith("["):
        assert value.endswith("]"), f"line {lineno}: unterminated inline list"
        parent[key] = [item.strip().strip('"\'') for item in value[1:-1].split(",") if item.strip()]
    else:
        if value.startswith('"'):
            assert value.endswith('"'), f"line {lineno}: unterminated quoted string"
            value = value[1:-1]
        parent[key] = value

assert root["name"] == "evolving-agent-process"
assert root["version"] == "0.1.0"
assert root["metadata"]["hermes"]["tags"] == ["dogfood", "process", "codex", "agent-development"]

required = [

    "Reuse scan: before adding helpers, serializers, coercion utilities, guards, adapters, or local wrappers",
    "search the same file and sibling modules for existing same-concern helpers or naming patterns",

]
assert not [item for item in required if item not in skill]

when_to_use = skill.split("## When to Use", 1)[1].split("Also use it", 1)[0]
